    with zipfile.ZipFile(path, 'w') as zp:
        zp.writestr("manifest.json", manifest_json)
        zp.writestr("bg.js", background_js)
    return path

def generate_links(count):
    """Generate multiple links with progress"""
    links = []
    success = 0
    fail = 0
    
    logger.info(f"🔄 Generating {count} links...")
    
    for i in range(count):
        logger.info(f"📌 Link {i+1}/{count}")
        link = generate_one_link()
        if link:
            links.append(link)
            success += 1
        else:
            fail += 1
        if i < count - 1:
            wait_time = random.uniform(10, 20)
            logger.info(f"⏳ Waiting {wait_time:.0f}s before next...")
            time.sleep(wait_time)
    
    logger.info(f"✅ Done! Success: {success}, Failed: {fail}")
    return links, success, fail

async def start(update: Update, context: CallbackContext):
    user = update.effective_user
    logger.info(f"📱 /start from user {user.id} ({user.username})")
    await update.message.reply_text(
        "🎵 <b>Spotify Link Generator</b>\n\n"
        "Send me a number (1-10) and I'll generate that many links.\n"
        "I'll send you a file with all the links.\n\n"
        "Example: <code>5</code>\n\n"
        "⚠️ Each link takes ~2 minutes to generate.",
        parse_mode='HTML'
    )

async def handle_message(update: Update, context: CallbackContext):
    user = update.effective_user
    text = update.message.text.strip()
    logger.info(f"📱 Message from user {user.id}: {text}")
    
    try:
        count = int(text)
        if count < 1 or count > 10:
            logger.warning(f"⚠️ User {user.id} sent invalid number: {count}")
            await update.message.reply_text("❌ Send a number between 1 and 10.")
            return
    except ValueError:
        logger.warning(f"⚠️ User {user.id} sent non-number: {text}")
        await update.message.reply_text("❌ Send a valid number (e.g., <code>5</code>)", parse_mode='HTML')
        return
    
    logger.info(f"✅ User {user.id} requested {count} links")
    
    status = await update.message.reply_text(
        f"🔄 Generating {count} links...\n"
        f"⏳ This will take ~{count*2} minutes.\n"
        f"📊 Check logs on VPS for progress."
    )
    
    # Generate links (this runs synchronously)
    links, success, fail = generate_links(count)
    
    if links:
        logger.info(f"📁 Generating file with {len(links)} links...")
        
        with tempfile.NamedTemporaryFile(mode='w', suffix='.txt', delete=False) as f:
            f.write(f"🎵 Spotify Links\n")
            f.write(f"Generated: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
            f.write("="*50 + "\n\n")
            for idx, link in enumerate(links, 1):
                f.write(f"{idx}. {link}\n")
            file_path = f.name
        
        with open(file_path, 'rb') as f:
            await update.message.reply_document(
                document=f,
                filename=f'spotify_links_{datetime.now().strftime("%Y%m%d_%H%M%S")}.txt',
                caption=f"✅ Generated {len(links)} links\n"
                        f"✅ Success: {success}\n"
                        f"❌ Failed: {fail}"
            )
        
        os.unlink(file_path)
        
        preview = "\n".join(links[:5])
        if len(links) > 5:
            preview += f"\n\n... and {len(links)-5} more in the file"
        await update.message.reply_text(f"🎵 <b>Your Links:</b>\n\n{preview}", parse_mode='HTML')
        logger.info(f"✅ Sent {len(links)} links to user {user.id}")
    else:
        logger.warning(f"❌ No links generated for user {user.id}")
        await update.message.reply_text("❌ Failed to generate any links. Try again later.")
    
    await status.delete()

async def cancel(update: Update, context: CallbackContext):
    user = update.effective_user
    logger.info(f"📱 /cancel from user {user.id}")
    await update.message.reply_text("❌ Cancelled. Send /start to restart.")

async def help_command(update: Update, context: CallbackContext):
    await update.message.reply_text(
        "🎵 <b>Spotify Link Generator</b>\n\n"
        "Send a number (1-10) to generate that many Spotify links.\n"
        "You'll receive a file with all the links.\n\n"
        "Commands:\n"
        "/start - Start the bot\n"
        "/help - Show this help\n"
        "/cancel - Cancel operation",
        parse_mode='HTML'
    )

def main():
    logger.info("🚀 Starting Spotify Link Generator Bot...")
    logger.info(f"Bot Token: {BOT_TOKEN[:10]}...")
    
    app = Application.builder().token(BOT_TOKEN).build()
    
    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("help", help_command))
    app.add_handler(CommandHandler("cancel", cancel))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_message))
    
    logger.info("✅ Bot is running! Send /start in Telegram.")
    logger.info("📋 Logs will appear here when users interact.")
    
    app.run_polling(allowed_updates=Update.ALL_TYPES)

if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        logger.info("🛑 Bot stopped.")
        sys.exit(0)
